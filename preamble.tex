% ---------- \boxenv{environment}{color} ----------
\newcommand{\boxenv}[2]{%
  \tcolorboxenvironment{#1}{
    enhanced jigsaw, breakable,
    colback=#2!6,
    colframe=#2!85!black,
    boxrule=0.8pt,
    arc=2pt,
    left=8pt, right=8pt, top=6pt, bottom=6pt,
    before skip=10pt, after skip=10pt
  }%
}

\boxenv{theorem}{thmcol}
\boxenv{lemma}{thmcol}
